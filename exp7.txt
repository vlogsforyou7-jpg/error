chatserver.c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <unistd.h>
#include <arpa/inet.h>
#include <pthread.h>

#define PORT 8080
#define MAX_CLIENTS 10
#define BUFFER_SIZE 1024

int clients[MAX_CLIENTS];
int client_count = 0;

pthread_mutex_t lock;

// Broadcast message to all clients except sender
void broadcast(char *message, int sender)
{
    pthread_mutex_lock(&lock);

    for (int i = 0; i < client_count; i++)
    {
        if (clients[i] != sender)
        {
            send(clients[i], message, strlen(message), 0);
        }
    }

    pthread_mutex_unlock(&lock);
}

// Handle each client
void *handle_client(void *arg)
{
    int client_socket = *(int *)arg;
    char buffer[BUFFER_SIZE];
    int len;

    while ((len = recv(client_socket, buffer, BUFFER_SIZE - 1, 0)) > 0)
    {
        buffer[len] = '\0';

        broadcast(buffer, client_socket);
    }

    close(client_socket);

    pthread_exit(NULL);
}

int main()
{
    int server_fd, client_socket;

    struct sockaddr_in server_addr, client_addr;
    socklen_t addr_size;

    // Initialize mutex
    pthread_mutex_init(&lock, NULL);

    // Create socket
    server_fd = socket(AF_INET, SOCK_STREAM, 0);

    // Configure server address
    server_addr.sin_family = AF_INET;
    server_addr.sin_port = htons(PORT);
    server_addr.sin_addr.s_addr = INADDR_ANY;

    // Bind socket
    bind(server_fd,
         (struct sockaddr *)&server_addr,
         sizeof(server_addr));

    // Listen for clients
    listen(server_fd, 5);

    printf("Server running on port %d\n", PORT);

    while (1)
    {
        addr_size = sizeof(client_addr);

        // Accept client
        client_socket = accept(server_fd,
                               (struct sockaddr *)&client_addr,
                               &addr_size);

        // Add client to array
        pthread_mutex_lock(&lock);

        if (client_count < MAX_CLIENTS)
        {
            clients[client_count++] = client_socket;
        }

        pthread_mutex_unlock(&lock);

        // Create thread for client
        pthread_t tid;

        pthread_create(&tid,
                       NULL,
                       handle_client,
                       &clients[client_count - 1]);

        pthread_detach(tid);
    }

    close(server_fd);

    pthread_mutex_destroy(&lock);

    return 0;
}





chatclient.c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <unistd.h>
#include <arpa/inet.h>
#include <pthread.h>

#define PORT 8080
#define BUFFER_SIZE 1024

int sockfd;

// Receive messages from server
void *receive_msg(void *arg)
{
    char message[BUFFER_SIZE + 1];
    int len;

    while ((len = recv(sockfd, message, BUFFER_SIZE, 0)) > 0)
    {
        message[len] = '\0';
        printf("%s", message);
        fflush(stdout);
    }

    return NULL;
}

int main()
{
    struct sockaddr_in server_addr;
    char message[BUFFER_SIZE];
    pthread_t tid;

    // Create socket
    sockfd = socket(AF_INET, SOCK_STREAM, 0);

    // Configure server address
    server_addr.sin_family = AF_INET;
    server_addr.sin_port = htons(PORT);
    inet_pton(AF_INET, "127.0.0.1", &server_addr.sin_addr);

    // Connect to server
    connect(sockfd,
            (struct sockaddr *)&server_addr,
            sizeof(server_addr));

    // Create thread to receive messages
    pthread_create(&tid, NULL, receive_msg, NULL);

    // Send messages to server
    while (fgets(message, BUFFER_SIZE, stdin))
    {
        send(sockfd, message, strlen(message), 0);
    }

    // Close socket
    close(sockfd);

    return 0;
}
